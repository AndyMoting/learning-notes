# core/config_manager.py
"""多层级配置管理器"""

import os
import yaml
from pathlib import Path
from typing import Any, Dict, Optional
from dataclasses import dataclass, field
import logging

logger = logging.getLogger(__name__)


@dataclass
class DatabaseConfig:
    host: str = "localhost"
    port: int = 5432
    name: str = "test_db"
    user: str = "test_user"
    password: str = ""
    pool_size: int = 5


@dataclass
class BrowserConfig:
    name: str = "chrome"
    headless: bool = False
    window_width: int = 1920
    window_height: int = 1080
    implicit_wait: int = 5
    page_load_timeout: int = 30


@dataclass
class FrameworkConfig:
    environment: str = "dev"
    parallel_workers: int = 4
    max_retries: int = 1
    screenshot_on_failure: bool = True
    log_level: str = "INFO"
    report_dir: str = "reports"
    data_dir: str = "data"


class ConfigManager:
    """
    配置管理器：支持 YAML 文件 + 环境变量 + 命令行参数
    
    优先级：命令行参数 > 环境变量 > 环境特定配置 > 主配置
    """
    
    _instance = None
    _config = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance
    
    def __init__(self):
        if self._config is None:
            self._config = self._load_config()
    
    def _load_config(self) -> Dict[str, Any]:
        """加载配置（多层合并）"""
        config_dir = Path(__file__).parent.parent / "config"
        
        # 1. 加载主配置
        main_config = self._load_yaml(config_dir / "settings.yaml")
        
        # 2. 加载环境特定配置（覆盖主配置）
        env = self._get_environment()
        env_file = config_dir / f"{env}.yaml"
        if env_file.exists():
            env_config = self._load_yaml(env_file)
            main_config = self._deep_merge(main_config, env_config)
            logger.info(f"已加载环境配置: {env}")
        
        # 3. 环境变量覆盖（前缀 TEST_）
        main_config = self._apply_env_overrides(main_config)
        
        return main_config
    
    def _load_yaml(self, path: Path) -> Dict:
        """加载 YAML 文件"""
        if not path.exists():
            logger.warning(f"配置文件不存在: {path}")
            return {}
        with open(path, "r", encoding="utf-8") as f:
            return yaml.safe_load(f) or {}
    
    def _get_environment(self) -> str:
        """获取当前环境名"""
        return os.environ.get("TEST_ENV", "dev")
    
    def _deep_merge(self, base: Dict, override: Dict) -> Dict:
        """深度合并两个字典"""
        result = base.copy()
        for key, value in override.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = self._deep_merge(result[key], value)
            else:
                result[key] = value
        return result
    
    def _apply_env_overrides(self, config: Dict) -> Dict:
        """应用环境变量覆盖（TEST_ 前缀）"""
        prefix = "TEST_"
        for key, value in os.environ.items():
            if key.startswith(prefix):
                # TEST_DB_HOST → db.host
                path = key[len(prefix):].lower().split("_")
                self._set_nested(config, path, self._cast_value(value))
        return config
    
    def _set_nested(self, d: Dict, path: list, value: Any):
        """设置嵌套字典值"""
        for key in path[:-1]:
            d = d.setdefault(key, {})
        d[path[-1]] = value
    
    @staticmethod
    def _cast_value(value: str) -> Any:
        """自动转换环境变量值为合适类型"""
        if value.lower() in ("true", "yes", "1"):
            return True
        if value.lower() in ("false", "no", "0"):
            return False
        try:
            return int(value)
        except ValueError:
            try:
                return float(value)
            except ValueError:
                return value
    
    def get(self, key_path: str, default: Any = None) -> Any:
        """
        通过点分隔路径获取配置值
        
        用法: config.get("db.host") → "localhost"
        """
        keys = key_path.split(".")
        value = self._config
        for key in keys:
            if isinstance(value, dict):
                value = value.get(key)
            else:
                return default
            if value is None:
                return default
        return value
    
    @property
    def db(self) -> DatabaseConfig:
        db_config = self._config.get("database", {})
        return DatabaseConfig(**db_config)
    
    @property
    def browser(self) -> BrowserConfig:
        browser_config = self._config.get("browser", {})
        return BrowserConfig(**browser_config)
    
    @property
    def framework(self) -> FrameworkConfig:
        fw_config = self._config.get("framework", {})
        return FrameworkConfig(**fw_config)
```

# core/logger.py
"""结构化日志管理器"""

import logging
import logging.handlers
import json
import sys
from pathlib import Path
from datetime import datetime
from typing import Optional


class StructuredFormatter(logging.Formatter):
    """JSON 结构化日志格式"""
    
    def format(self, record: logging.LogRecord) -> str:
        log_entry = {
            "timestamp": datetime.utcnow().isoformat() + "Z",
            "level": record.levelname,
            "logger": record.name,
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno,
            "message": record.getMessage(),
        }
        
        # 添加额外字段
        if hasattr(record, "test_case"):
            log_entry["test_case"] = record.test_case
        if hasattr(record, "step"):
            log_entry["step"] = record.step
        if hasattr(record, "duration_ms"):
            log_entry["duration_ms"] = record.duration_ms
        
        # 异常信息
        if record.exc_info and record.exc_info[0]:
            log_entry["exception"] = {
                "type": record.exc_info[0].__name__,
                "message": str(record.exc_info[1]),
                "traceback": self.formatException(record.exc_info)
            }
        
        return json.dumps(log_entry, ensure_ascii=False)


class TestAdapter(logging.LoggerAdapter):
    """带测试上下文的日志适配器"""
    
    def process(self, msg, kwargs):
        extra = kwargs.get("extra", {})
        extra.update(self.extra)
        kwargs["extra"] = extra
        return msg, kwargs
    
    def step(self, step_name: str, **kwargs):
        """记录测试步骤"""
        self.info(f"[STEP] {step_name}", extra={"step": step_name, **kwargs})
    
    def assertion(self, description: str, passed: bool, **kwargs):
        """记录断言结果"""
        status = "PASS" if passed else "FAIL"
        level = logging.INFO if passed else logging.ERROR
        self.log(level, f"[ASSERT-{status}] {description}", 
                 extra={"assertion": description, "passed": passed, **kwargs})


def setup_logging(
    log_dir: str = "reports",
    log_level: str = "INFO",
    enable_console: bool = True,
    enable_file: bool = True
) -> None:
    """初始化日志系统"""
    
    log_path = Path(log_dir)
    log_path.mkdir(parents=True, exist_ok=True)
    
    root_logger = logging.getLogger()
    root_logger.setLevel(getattr(logging, log_level.upper()))
    
    formatter = StructuredFormatter()
    
    if enable_console:
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.setFormatter(formatter)
        root_logger.addHandler(console_handler)
    
    if enable_file:
        file_handler = logging.handlers.RotatingFileHandler(
            filename=log_path / "test_execution.log",
            maxBytes=10 * 1024 * 1024,  # 10MB
            backupCount=5,
            encoding="utf-8"
        )
        file_handler.setFormatter(formatter)
        root_logger.addHandler(file_handler)
        
        # 错误日志单独文件
        error_handler = logging.handlers.RotatingFileHandler(
            filename=log_path / "test_errors.log",
            maxBytes=5 * 1024 * 1024,
            backupCount=3,
            encoding="utf-8"
        )
        error_handler.setLevel(logging.ERROR)
        error_handler.setFormatter(formatter)
        root_logger.addHandler(error_handler)


def get_logger(name: str, **context) -> TestAdapter:
    """获取带上下文的日志器"""
    logger = logging.getLogger(name)
    return TestAdapter(logger, context)

"""测试报告管理器 - Allure 集成"""

import allure
import json
from pathlib import Path
from typing import Optional, Dict, Any
from datetime import datetime


class ReportManager:
    """Allure 报告管理器"""
    
    def __init__(self, report_dir: str = "reports"):
        self.report_dir = Path(report_dir)
        self.report_dir.mkdir(parents=True, exist_ok=True)
    
    @staticmethod
    def attach_screenshot(driver, name: str = "screenshot"):
        """附加截图到报告"""
        allure.attach(
            driver.get_screenshot_as_png(),
            name=name,
            attachment_type=allure.attachment_type.PNG
        )
    
    @staticmethod
    def attach_page_source(driver, name: str = "page_source"):
        """附加页面源码"""
        allure.attach(
            driver.page_source,
            name=name,
            attachment_type=allure.attachment_type.HTML
        )
    
    @staticmethod
    def attach_json(data: Dict, name: str = "response_data"):
        """附加 JSON 数据"""
        allure.attach(
            json.dumps(data, indent=2, ensure_ascii=False),
            name=name,
            attachment_type=allure.attachment_type.JSON
        )
    
    @staticmethod
    def attach_log(log_content: str, name: str = "execution_log"):
        """附加日志内容"""
        allure.attach(
            log_content,
            name=name,
            attachment_type=allure.attachment_type.TEXT
        )
    
    @staticmethod
    def set_test_category(feature: str, story: str, severity: str = "normal"):
        """设置测试分类"""
        allure.dynamic.feature(feature)
        allure.dynamic.story(story)
        allure.dynamic.severity(severity)
    
    @staticmethod
    def add_link(url: str, name: str = None, link_type: str = "issue"):
        """添加外部链接（如 JIRA issue）"""
        allure.dynamic.link(url, name=name, type=link_type)

# core/plugin_manager.py
"""插件管理器 - 支持运行时加载测试插件"""

import importlib
import logging
from abc import ABC, abstractmethod
from typing import Dict, List, Type, Any
from pathlib import Path

logger = logging.getLogger(__name__)


class TestPlugin(ABC):
    """测试插件基类"""
    
    name: str = "base_plugin"
    version: str = "1.0.0"
    
    @abstractmethod
    def setup(self, config: Dict[str, Any]):
        """插件初始化"""
        pass
    
    @abstractmethod
    def on_test_start(self, test_name: str, test_data: Dict = None):
        """测试开始时调用"""
        pass
    
    @abstractmethod
    def on_test_end(self, test_name: str, result: str, duration: float):
        """测试结束时调用"""
        pass
    
    @abstractmethod
    def teardown(self):
        """插件销毁"""
        pass


class PluginManager:
    """插件管理器"""
    
    def __init__(self):
        self._plugins: Dict[str, TestPlugin] = {}
    
    def register(self, plugin: TestPlugin, config: Dict[str, Any] = None):
        """注册插件"""
        plugin.setup(config or {})
        self._plugins[plugin.name] = plugin
        logger.info(f"插件已注册: {plugin.name} v{plugin.version}")
    
    def unregister(self, plugin_name: str):
        """注销插件"""
        if plugin_name in self._plugins:
            self._plugins[plugin_name].teardown()
            del self._plugins[plugin_name]
            logger.info(f"插件已注销: {plugin_name}")
    
    def load_from_directory(self, plugin_dir: str):
        """从目录自动加载插件"""
        plugin_path = Path(plugin_dir)
        if not plugin_path.exists():
            logger.warning(f"插件目录不存在: {plugin_dir}")
            return
        
        for file in plugin_path.glob("*/plugin.py"):
            module_name = f"plugins.{file.parent.name}.plugin"
            try:
                module = importlib.import_module(module_name)
                for attr_name in dir(module):
                    attr = getattr(module, attr_name)
                    if (isinstance(attr, type) and 
                        issubclass(attr, TestPlugin) and 
                        attr is not TestPlugin):
                        self.register(attr())
            except Exception as e:
                logger.error(f"加载插件失败 {module_name}: {e}")
    
    def notify_test_start(self, test_name: str, test_data: Dict = None):
        """通知所有插件：测试开始"""
        for plugin in self._plugins.values():
            try:
                plugin.on_test_start(test_name, test_data)
            except Exception as e:
                logger.error(f"插件 {plugin.name} on_test_start 失败: {e}")
    
    def notify_test_end(self, test_name: str, result: str, duration: float):
        """通知所有插件：测试结束"""
        for plugin in self._plugins.values():
            try:
                plugin.on_test_end(test_name, result, duration)
            except Exception as e:
                logger.error(f"插件 {plugin.name} on_test_end 失败: {e}")
    
    def shutdown(self):
        """关闭所有插件"""
        for plugin in self._plugins.values():
            plugin.teardown()
        self._plugins.clear()

        pass

                "description": f"自动化测试用例 {test_name} 在 {datetime.now()} 执行失败",
                "issuetype": {"name": "Bug"},
                "priority": {"name": "High"}
            }
        }
        try:
            requests.post(url, json=payload, auth=self.api_token)
        except Exception as e:
            logger.error(f"JIRA 创建缺陷失败: {e}")
    
    def teardown(self):

"""测试基类 - 所有测试用例的父类"""

import pytest
import logging
from typing import Optional
from core.config_manager import ConfigManager
from core.report_manager import ReportManager
from core.logger import get_logger
from utils.driver_factory import DriverManager


class BaseTest:
    """
    测试基类
    
    提供：
    - 配置访问
    - 日志记录（带测试上下文）
    - 报告附件
    - 通用断言辅助
    """
    
    config = ConfigManager()
    
    @pytest.fixture(autouse=True)
    def setup_test_context(self, request):
        """自动注入测试上下文"""
        self.test_name = request.node.name
        self.logger = get_logger(
            f"test.{request.node.module.__name__}",
            test_case=self.test_name
        )
        self.report = ReportManager()
        
        self.logger.step(f"测试开始: {self.test_name}")
        self._start_time = __import__("time").perf_counter()
        
        yield
        
        duration = __import__("time").perf_counter() - self._start_time
        self.logger.step(f"测试结束: {self.test_name} | 耗时: {duration:.2f}s")
    
    def assert_with_screenshot(self, condition: bool, message: str, driver=None):
        """带截图的断言"""
        self.logger.assertion(message, condition)
        if not condition:
            if driver:
                self.report.attach_screenshot(driver, f"断言失败_{self.test_name}")
            assert condition, message
    
    def assert_response_status(self, response, expected_status: int):
        """断言 HTTP 响应状态码"""
        actual = response.status_code
        self.logger.assertion(
            f"状态码验证: 期望 {expected_status}, 实际 {actual}",
            actual == expected_status
        )
        assert actual == expected_status, (
            f"期望状态码 {expected_status}，实际 {actual}，"
            f"响应体: {response.text[:500]}"
        )
    
    def assert_response_contains(self, response, expected_key: str, expected_value=None):
        """断言响应体包含指定字段"""
        data = response.json()
        self.logger.assertion(
            f"响应包含字段: {expected_key}",
            expected_key in data
        )
        assert expected_key in data, f"响应中缺少字段: {expected_key}"
        
        if expected_value is not None:
            actual = data[expected_key]
            self.logger.assertion(
                f"字段 {expected_key} 值验证: 期望 {expected_value}, 实际 {actual}",
                actual == expected_value
            )
            assert actual == expected_value
```

## 常见坑

1. **配置硬编码**：URL、账号等散落在测试代码中——统一到配置管理器
2. **日志无结构**：纯文本日志无法被 ELK/Splunk 解析——使用 JSON 结构化日志
3. **框架与工具耦合**：直接依赖 pytest 导致无法切换到 unittest——抽象测试运行层
4. **报告信息不足**：只有通过/失败，缺少上下文——自动附加截图、日志、请求响应
5. **插件加载顺序问题**：插件间有依赖关系——定义优先级和依赖声明
