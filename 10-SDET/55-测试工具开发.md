## 核心概念

### 测试工具分类

| 类型 | 用途 | 技术栈 |
|------|------|--------|
| CLI 工具 | 命令行执行测试任务 | argparse / click |
| 报告生成器 | 将原始结果转为可读报告 | Jinja2 / WeasyPrint |
| Mock Server | 模拟外部服务响应 | Flask / FastAPI |
| 数据工厂 | 批量生成测试数据 | Faker / 自定义生成器 |
| 结果分析器 | 分析测试历史趋势 | pandas / matplotlib |

# tools/test_cli.py
"""测试执行 CLI 工具"""

import click
import subprocess
import sys
import json
from pathlib import Path
from datetime import datetime
from typing import Optional


@click.group()
@click.version_option(version="1.0.0")
def cli():
    """测试管理工具集 - 统一测试执行入口"""
    pass


@cli.command()
@click.option("--env", "-e", default="dev", 
              type=click.Choice(["dev", "staging", "production"]),
              help="测试环境")
@click.option("--browser", "-b", default="chrome",
              type=click.Choice(["chrome", "firefox", "edge"]),
              help="浏览器类型")
@click.option("--headless/--no-headless", default=False,
              help="无头模式")
@click.option("--markers", "-m", multiple=True,
              help="pytest marker 过滤（可多次指定）")
@click.option("--parallel", "-p", default=1, type=int,
              help="并行工作进程数")
@click.option("--report-format", "-r", 
              type=click.Choice(["allure", "html", "json"]),
              default="allure",
              help="报告格式")
@click.option("--output-dir", "-o", default="reports",
              help="报告输出目录")
def run(env, browser, headless, markers, parallel, report_format, output_dir):
    """执行测试套件"""
    
    # 构建 pytest 参数
    pytest_args = ["pytest"]
    
    # 环境配置
    pytest_args.extend(["--env", env])
    pytest_args.extend(["--browser", browser])
    if headless:
        pytest_args.append("--headless")
    
    # Marker 过滤
    if markers:
        marker_expr = " or ".join(markers)
        pytest_args.extend(["-m", marker_expr])
    
    # 并行执行
    if parallel > 1:
        pytest_args.extend(["-n", str(parallel)])
    
    # 报告配置
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    report_path = Path(output_dir) / timestamp
    report_path.mkdir(parents=True, exist_ok=True)
    
    if report_format == "allure":
        pytest_args.extend(["--alluredir", str(report_path / "allure-results")])
    elif report_format == "html":
        pytest_args.extend(["--html", str(report_path / "report.html"), "--self-contained-html"])
    elif report_format == "json":
        pytest_args.extend(["--json-report", "--json-report-file", str(report_path / "report.json")])
    
    # 执行
    click.echo(f"执行命令: {' '.join(pytest_args)}")
    click.echo(f"环境: {env} | 浏览器: {browser} | 并行: {parallel}")
    click.echo("-" * 60)
    
    result = subprocess.run(pytest_args)
    
    click.echo("-" * 60)
    if result.returncode == 0:
        click.echo(click.style("✓ 全部通过", fg="green", bold=True))
    else:
        click.echo(click.style(f"✗ 存在失败 (exit code: {result.returncode})", fg="red", bold=True))
    
    click.echo(f"报告路径: {report_path}")
    sys.exit(result.returncode)


@cli.command()
@click.argument("result_dir", type=click.Path(exists=True))
@click.option("--port", "-p", default=8080, help="报告服务端口")
def serve(result_dir, port):
    """启动报告查看服务"""
    import http.server
    import socketserver
    
    click.echo(f"启动报告服务: http://localhost:{port}")
    click.echo(f"报告目录: {result_dir}")
    
    handler = http.server.SimpleHTTPRequestHandler
    with socketserver.TCPServer(("", port), handler) as httpd:
        httpd.serve_forever()


@cli.command()
@click.option("--count", "-c", default=100, help="生成数据条数")
@click.option("--type", "-t", 
              type=click.Choice(["user", "order", "product", "all"]),
              default="user",
              help="数据类型")
@click.option("--output", "-o", default="data/generated_data.json",
              help="输出文件路径")
@click.option("--locale", "-l", default="zh_CN", help="数据语言区域")
def generate(count, type, output, locale):
    """生成测试数据"""
    from factories.user_factory import UserFactory
    from factories.order_factory import OrderFactory
    from factories.product_factory import ProductFactory
    
    data = {}
    
    if type in ("user", "all"):
        data["users"] = [UserFactory.create() for _ in range(count)]
    if type in ("order", "all"):
        data["orders"] = [OrderFactory.create() for _ in range(count)]
    if type in ("product", "all"):
        data["products"] = [ProductFactory.create() for _ in range(count)]
    
    output_path = Path(output)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(data, f, default=str, ensure_ascii=False, indent=2)
    
    click.echo(f"已生成 {count} 条 {type} 数据 → {output_path}")


@cli.command()
@click.argument("report_path", type=click.Path(exists=True))
def analyze(report_path):
    """分析测试报告"""
    with open(report_path, "r", encoding="utf-8") as f:
        data = json.load(f)
    
    summary = data.get("summary", {})
    
    click.echo("=" * 50)
    click.echo("测试执行分析报告")
    click.echo("=" * 50)
    click.echo(f"总用例数: {summary.get('total', 0)}")
    click.echo(click.style(f"通过: {summary.get('passed', 0)}", fg="green"))
    click.echo(click.style(f"失败: {summary.get('failed', 0)}", fg="red"))
    click.echo(click.style(f"跳过: {summary.get('skipped', 0)}", fg="yellow"))
    click.echo(f"通过率: {summary.get('passed', 0) / max(summary.get('total', 1), 1) * 100:.1f}%")
    click.echo(f"总耗时: {summary.get('duration', 0):.1f}s")
    
    # 失败用例详情
    if summary.get("failed", 0) > 0:
        click.echo("\n失败用例:")
        for test in data.get("tests", []):
            if test.get("outcome") == "failed":
                click.echo(click.style(f"  ✗ {test['nodeid']}", fg="red"))


if __name__ == "__main__":
    cli()

                <tr>
                    <td>{{ test.name }}</td>
                    <td class="status-{{ test.outcome }}">{{ test.outcome | upper }}</td>

                    <td>{{ test.error or "-" }}</td>
                </tr>
                {% endfor %}
            </tbody>
        </table>
        
        <div class="footer">
            由测试框架自动生成 | 通过率: {{ "%.1f"|format(summary.passed / summary.total * 100) if summary.total > 0 else 0 }}%
        </div>
    </div>
</body>
</html>
"""


class ReportGenerator:
    """测试报告生成器"""
    
    def __init__(self, template_dir: str = None):
        if template_dir:
            self.env = Environment(loader=FileSystemLoader(template_dir))
        else:
            self.env = Environment()
    
    def generate_html(self, result_data: Dict, output_path: str) -> str:
        """从 JSON 结果生成 HTML 报告"""
        template = self.env.from_string(HTML_TEMPLATE)
        
        html_content = template.render(
            summary=result_data.get("summary", {}),
            tests=result_data.get("tests", []),
            environment=result_data.get("environment", "unknown"),
            report_time=datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        )
        
        output = Path(output_path)
        output.parent.mkdir(parents=True, exist_ok=True)
        output.write_text(html_content, encoding="utf-8")
        
        return str(output)
    
    def generate_pdf(self, html_path: str, output_path: str) -> str:
        """将 HTML 报告转为 PDF"""
        try:
            from weasyprint import HTML
            HTML(filename=html_path).write_pdf(output_path)
            return output_path
        except ImportError:
            raise ImportError("请安装 weasyprint: pip install weasyprint")
    
    def generate_from_json(self, json_path: str, output_dir: str) -> Dict[str, str]:
        """从 JSON 文件生成完整报告"""
        with open(json_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        
        output_path = Path(output_dir)
        output_path.mkdir(parents=True, exist_ok=True)
        
        # 生成 HTML
        html_path = output_path / "report.html"
        self.generate_html(data, str(html_path))
        
        # 生成 PDF
        pdf_path = output_path / "report.pdf"
        try:
            self.generate_pdf(str(html_path), str(pdf_path))
        except Exception:
            pdf_path = None
        
        return {
            "html": str(html_path),
            "pdf": str(pdf_path) if pdf_path else None
        }


# 使用示例
if __name__ == "__main__":

    # 模拟测试结果

# tools/mock_server.py
"""Mock Server - 模拟外部服务接口"""

from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse
from pydantic import BaseModel
from typing import Dict, List, Optional
import uvicorn
import json
from pathlib import Path


app = FastAPI(title="测试 Mock Server", version="1.0.0")

# 内存数据存储
mock_data = {
    "users": {},
    "orders": {},
    "products": {}
}

# 请求日志
request_log: List[Dict] = []


@app.middleware("http")
async def log_requests(request: Request, call_next):
    """记录所有请求"""
    response = await call_next(request)
    request_log.append({
        "method": request.method,
        "path": request.url.path,
        "status": response.status_code,
        "timestamp": __import__("datetime").datetime.now().isoformat()
    })
    return response


# ============ 用户服务 Mock ============

class UserCreate(BaseModel):
    username: str
    email: str
    password: str


@app.post("/api/v1/users")
def create_user(user: UserCreate):
    """创建用户"""
    if user.email in mock_data["users"]:
        raise HTTPException(status_code=409, detail="邮箱已存在")
    
    user_id = str(len(mock_data["users"]) + 1)
    mock_data["users"][user.email] = {
        "id": user_id,
        "username": user.username,
        "email": user.email,
        "created_at": "2026-07-17T10:00:00Z"
    }
    return mock_data["users"][user.email]


@app.get("/api/v1/users/{user_id}")
def get_user(user_id: str):
    """获取用户信息"""
    for user in mock_data["users"].values():
        if user["id"] == user_id:
            return user
    raise HTTPException(status_code=404, detail="用户不存在")


@app.delete("/api/v1/users/{user_id}")
def delete_user(user_id: str):
    """删除用户"""
    for email, user in mock_data["users"].items():
        if user["id"] == user_id:
            del mock_data["users"][email]
            return {"message": "删除成功"}
    raise HTTPException(status_code=404, detail="用户不存在")

# tools/data_factory.py
"""测试数据工厂工具 - 批量生成各类测试数据"""

from faker import Faker
from typing import List, Dict, Any, Optional
from dataclasses import dataclass, asdict
import json
import csv
from pathlib import Path


class DataFactory:
    """通用数据工厂"""
    
    def __init__(self, locale: str = "zh_CN"):
        self.fake = Faker(locale)
        self.fake.seed_instance(42)  # 可重现
    
    def generate_users(self, count: int, **overrides) -> List[Dict]:
        """生成用户数据"""
        users = []
        for _ in range(count):
            user = {
                "id": self.fake.uuid4()[:8],
                "username": self.fake.user_name(),
                "email": self.fake.email(),
                "phone": self.fake.phone_number(),
                "address": self.fake.address(),
                "company": self.fake.company(),
                "birthdate": self.fake.date_of_birth(minimum_age=18, maximum_age=65).isoformat(),
                "is_active": self.fake.boolean(chance_of_getting_true=80),
                "created_at": self.fake.date_time_this_year().isoformat()
            }
            user.update(overrides)
            users.append(user)
        return users
    
    def generate_products(self, count: int, **overrides) -> List[Dict]:
        """生成商品数据"""
        products = []
        categories = ["电子产品", "服装", "食品", "家居", "图书"]
        for _ in range(count):
            product = {
                "id": self.fake.uuid4()[:8],
                "name": self.fake.catch_phrase(),
                "category": self.fake.random_element(categories),
                "price": round(self.fake.pydecimal(min_value=10, max_value=9999, right_digits=2), 2),
                "stock": self.fake.random_int(min=0, max=1000),
                "sku": self.fake.ean13(),
                "description": self.fake.text(max_nb_chars=200),
                "is_available": self.fake.boolean(chance_of_getting_true=90)
            }
            product.update(overrides)
            products.append(product)
        return products
    
    def generate_orders(self, count: int, user_ids: List[str] = None, **overrides) -> List[Dict]:
        """生成订单数据"""
        orders = []
        statuses = ["pending", "paid", "shipped", "delivered", "cancelled"]
        weights = [20, 30, 25, 20, 5]
        
        for _ in range(count):
            order = {
                "id": f"ORD-{self.fake.random_int(10000, 99999)}",
                "user_id": self.fake.random_element(user_ids) if user_ids else self.fake.uuid4()[:8],
                "status": self.fake.random_elements(statuses, length=1, weights=weights)[0],
                "total_amount": round(self.fake.pydecimal(min_value=50, max_value=5000, right_digits=2), 2),
                "items_count": self.fake.random_int(min=1, max=10),
                "created_at": self.fake.date_time_this_year().isoformat()
            }
            order.update(overrides)
            orders.append(order)
        return orders
    
    def generate_test_matrix(self, dimensions: Dict[str, List]) -> List[Dict]:
        """
        生成测试矩阵（笛卡尔积）
        
        用法:
            factory.generate_test_matrix({
                "browser": ["chrome", "firefox"],
                "os": ["windows", "macos"],
                "resolution": ["1920x1080", "1366x768"]
            })
        """
        from itertools import product
        
        keys = list(dimensions.keys())
        values = list(dimensions.values())
        
        matrix = []
        for combo in product(*values):
            matrix.append(dict(zip(keys, combo)))
        
        return matrix
    
    @staticmethod
    def export_json(data: List[Dict], output_path: str):
        """导出为 JSON"""
        path = Path(output_path)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
    
    @staticmethod
    def export_csv(data: List[Dict], output_path: str):
        """导出为 CSV"""
        if not data:
            return
        path = Path(output_path)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8", newline="") as f:
            writer = csv.DictWriter(f, fieldnames=data[0].keys())
            writer.writeheader()
            writer.writerows(data)


# 使用示例
if __name__ == "__main__":
    factory = DataFactory("zh_CN")
    
    # 生成用户数据
    users = factory.generate_users(100)
    DataFactory.export_json(users, "data/users.json")
    DataFactory.export_csv(users, "data/users.csv")
    
    # 生成商品数据
    products = factory.generate_products(50)
    DataFactory.export_json(products, "data/products.json")
    
    # 生成订单数据
    user_ids = [u["id"] for u in users[:20]]
    orders = factory.generate_orders(200, user_ids=user_ids)
    DataFactory.export_json(orders, "data/orders.json")
    
    # 生成测试矩阵
    matrix = factory.generate_test_matrix({
        "browser": ["chrome", "firefox", "edge"],
        "os": ["windows", "macos", "linux"],
        "language": ["zh_CN", "en_US"]
    })
    print(f"测试矩阵: {len(matrix)} 种组合")
    DataFactory.export_json(matrix, "data/test_matrix.json")

## 常见坑

1. **CLI 参数爆炸**：工具功能增长导致参数过多——使用子命令分组
2. **Mock 与真实服务脱节**：Mock 响应与真实 API 差异大导致测试失效——定期对比契约
3. **数据工厂不可重现**：随机数据导致测试失败无法复现——使用固定 seed
4. **报告模板硬编码**：样式变更需要改代码——使用 Jinja2 模板分离
5. **Mock Server 无状态隔离**：测试间数据互相污染——提供 reset 接口
