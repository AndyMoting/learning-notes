### 实操1：Page Object Model（POM）

Page Object 是 UI 自动化测试中最核心的设计模式。每个页面对应一个类，封装页面元素定位和交互操作。

```
项目结构：
pages/
├── base_page.py          # 基类，封装通用操作
├── login_page.py         # 登录页
├── product_list_page.py  # 商品列表页
├── cart_page.py.py       # 购物车页
└── checkout_page.py      # 结算页
```

```python
# pages/base_page.py
"""所有 Page Object 的基类"""

from selenium.webdriver.remote.webdriver import WebDriver
from selenium.webdriver.remote.webelement import WebElement
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import TimeoutException, NoSuchElementException
import logging

logger = logging.getLogger(__name__)


class BasePage:
    """页面对象基类"""
    
    DEFAULT_TIMEOUT = 10
    POLL_FREQUENCY = 0.5
    
    def __init__(self, driver: WebDriver):
        self.driver = driver
        self.wait = WebDriverWait(
            driver, 
            self.DEFAULT_TIMEOUT,
            poll_frequency=self.POLL_FREQUENCY
        )
    
    def find_element(self, locator: tuple) -> WebElement:
        """查找单个元素（显式等待）"""
        try:
            element = self.wait.until(EC.presence_of_element_located(locator))
            logger.debug(f"找到元素: {locator}")
            return element
        except TimeoutException:
            logger.error(f"元素未找到: {locator}")
            self._take_screenshot("element_not_found")
            raise
    
    def find_clickable_element(self, locator: tuple) -> WebElement:
        """查找可点击元素"""
        try:
            return self.wait.until(EC.element_to_be_clickable(locator))
        except TimeoutException:
            logger.error(f"元素不可点击: {locator}")
            raise
    
    def click(self, locator: tuple):
        """点击元素"""
        element = self.find_clickable_element(locator)
        element.click()
        logger.info(f"点击元素: {locator}")
    
    def input_text(self, locator: tuple, text: str):
        """输入文本"""
        element = self.find_element(locator)
        element.clear()
        element.send_keys(text)
        logger.info(f"在 {locator} 输入: {text}")
    
    def get_text(self, locator: tuple) -> str:
        """获取元素文本"""
        return self.find_element(locator).text
    
    def is_element_visible(self, locator: tuple) -> bool:
        """判断元素是否可见"""
        try:
            self.wait.until(EC.visibility_of_element_located(locator))
            return True
        except TimeoutException:
            return False
    
    def wait_for_url_contains(self, url_part: str):
        """等待 URL 包含指定字符串"""
        self.wait.until(EC.url_contains(url_part))
    
    def _take_screenshot(self, name: str):
        """截图（供失败时调用）"""
        path = f"reports/screenshots/{name}_{self.__class__.__name__}.png"
        self.driver.save_screenshot(path)
        logger.info(f"截图已保存: {path}")
```

```python
"""登录页 Page Object"""

from selenium.webdriver.common.by import By
from pages.base_page import BasePage


class LoginPage(BasePage):
    """登录页面"""
    
    # 元素定位器
    USERNAME_INPUT = (By.ID, "username")
    PASSWORD_INPUT = (By.ID, "password")
    LOGIN_BUTTON = (By.CSS_SELECTOR, "button[type='submit']")
    ERROR_MESSAGE = (By.CLASS_NAME, "error-message")
    FORGOT_PASSWORD_LINK = (By.LINK_TEXT, "忘记密码")
    
    # 页面 URL
    URL = "/login"
    
    def navigate_to(self):
        """导航到登录页"""
        self.driver.get(self.URL)
        logger.info("导航到登录页")
        return self
    
    def login(self, username: str, password: str) -> 'LoginPage':
        """执行登录操作"""
        self.input_text(self.USERNAME_INPUT, username)
        self.input_text(self.PASSWORD_INPUT, password)
        self.click(self.LOGIN_BUTTON)
        logger.info(f"登录操作: username={username}")
        return self
    
    def get_error_message(self) -> str:
        """获取错误提示"""
        return self.get_text(self.ERROR_MESSAGE)
    
    def is_login_button_enabled(self) -> bool:
        """判断登录按钮是否可用"""
        return self.find_element(self.LOGIN_BUTTON).is_enabled()
    
    def click_forgot_password(self):
        """点击忘记密码"""
        self.click(self.FORGOT_PASSWORD_LINK)
```

```python
"""购物车页 Page Object"""

from selenium.webdriver.common.by import By
from pages.base_page import BasePage
from typing import List


class CartPage(BasePage):
    """购物车页面"""
    
    CART_ITEMS = (By.CSS_SELECTOR, ".cart-item")
    ITEM_NAME = (By.CSS_SELECTOR, ".item-name")
    ITEM_PRICE = (By.CSS_SELECTOR, ".item-price")
    ITEM_QUANTITY = (By.CSS_SELECTOR, ".item-quantity input")
    REMOVE_BUTTON = (By.CSS_SELECTOR, ".remove-item")
    CHECKOUT_BUTTON = (By.ID, "checkout")
    TOTAL_AMOUNT = (By.ID, "total-amount")
    EMPTY_CART_MESSAGE = (By.CLASS_NAME, "empty-cart")
    
    def get_cart_items_count(self) -> int:
        """获取购物车商品数量"""
        items = self.driver.find_elements(*self.CART_ITEMS)
        return len(items)
    
    def get_item_names(self) -> List[str]:
        """获取所有商品名称"""
        items = self.driver.find_elements(*self.ITEM_NAME)
        return [item.text for item in items]
    
    def update_quantity(self, item_index: int, quantity: int):
        """更新商品数量"""
        items = self.driver.find_elements(*self.ITEM_QUANTITY)
        items[item_index].clear()
        items[item_index].send_keys(str(quantity))
    
    def remove_item(self, item_index: int):
        """删除指定商品"""
        items = self.driver.find_elements(*self.REMOVE_BUTTON)
        items[item_index].click()
    
    def get_total_amount(self) -> str:
        """获取总金额"""
        return self.get_text(self.TOTAL_AMOUNT)
    
    def proceed_to_checkout(self):
        """进入结算"""
        self.click(self.CHECKOUT_BUTTON)
    
    def is_cart_empty(self) -> bool:
        """判断购物车是否为空"""
        return self.is_element_visible(self.EMPTY_CART_MESSAGE)
```

```python
# factories/user_factory.py
"""用户测试数据工厂"""

from dataclasses import dataclass, field
from typing import Optional, List
from faker import Faker
import uuid

fake = Faker("zh_CN")  # 中文数据


@dataclass
class User:
    username: str
    email: str
    password: str
    phone: str
    address: str = ""
    is_active: bool = True
    user_id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    role: str = "user"


class UserFactory:
    """用户数据工厂 - 生成各种测试用户数据"""
    
    @staticmethod
    def create(**overrides) -> User:
        """创建一个用户，允许覆盖任意字段"""
        defaults = {
            "username": fake.user_name(),
            "email": fake.email(),
            "password": "Test@123456",
            "phone": fake.phone_number(),
            "address": fake.address(),
            "is_active": True,
            "role": "user"
        }
        defaults.update(overrides)
        return User(**defaults)
    
    @staticmethod
    def create_admin(**overrides) -> User:
        """创建管理员用户"""
        overrides["role"] = "admin"
        return UserFactory.create(**overrides)
    
    @staticmethod
    def create_batch(count: int, **common_attrs) -> List[User]:
        """批量创建用户"""
        return [UserFactory.create(**common_attrs) for _ in range(count)]
    
    @staticmethod
    def create_invalid_users() -> dict:
        """创建各种无效用户数据（用于边界测试）"""
        base = UserFactory.create()
        return {
            "empty_username": UserFactory.create(username=""),
            "short_password": UserFactory.create(password="123"),
            "invalid_email": UserFactory.create(email="not-an-email"),
            "long_username": UserFactory.create(username="a" * 256),
            "special_chars": UserFactory.create(username="<script>alert(1)</script>"),
            "duplicate_email": UserFactory.create(email=base.email),  # 用于重复测试
        }


# 使用示例
if __name__ == "__main__":
    # 创建单个用户
    user = UserFactory.create()
    print(f"普通用户: {user}")
    
    # 创建管理员
    admin = UserFactory.create_admin(username="admin_test")
    print(f"管理员: {admin}")
    
    # 批量创建
    users = UserFactory.create_batch(10, is_active=True)
    print(f"批量创建: {len(users)} 个用户")
    
    # 无效数据
    invalid = UserFactory.create_invalid_users()
```

```python
# utils/driver_factory.py
"""WebDriver 单例管理器"""

from selenium import webdriver
from selenium.webdriver.chrome.options import Options as ChromeOptions
from selenium.webdriver.firefox.options import Options as FirefoxOptions
from typing import Optional
import threading


class DriverManager:
    """
    WebDriver 单例管理器
    保证同一线程只有一个 driver 实例
    """
    
    _instance: Optional[webdriver.Remote] = None
    _lock = threading.Lock()
    
    @classmethod
    def get_driver(cls, browser: str = "chrome", headless: bool = False) -> webdriver.Remote:
        """获取 WebDriver 实例（单例）"""
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = cls._create_driver(browser, headless)
        return cls._instance
    
    @classmethod
    def _create_driver(cls, browser: str, headless: bool) -> webdriver.Remote:
        """创建指定浏览器的 driver"""
        browser = browser.lower()
        
        if browser == "chrome":
            options = ChromeOptions()
            if headless:
                options.add_argument("--headless=new")
            options.add_argument("--no-sandbox")
            options.add_argument("--disable-dev-shm-usage")
            options.add_argument("--window-size=1920,1080")
            driver = webdriver.Chrome(options=options)
            
        elif browser == "firefox":
            options = FirefoxOptions()
            if headless:
                options.add_argument("--headless")
            driver = webdriver.Firefox(options=options)
            
        elif browser == "edge":
            from selenium.webdriver.edge.options import Options as EdgeOptions
            options = EdgeOptions()
            if headless:
                options.add_argument("--headless=new")
            driver = webdriver.Edge(options=options)
        else:
            raise ValueError(f"不支持的浏览器: {browser}")
        
        driver.implicitly_wait(5)
        driver.maximize_window()
        return driver
    
    @classmethod
    def quit_driver(cls):
        """销毁 driver 实例"""
        if cls._instance:
            cls._instance.quit()
            cls._instance = None
    
    @classmethod
    def restart_driver(cls, browser: str = "chrome", headless: bool = False):
        """重启 driver"""
        cls.quit_driver()
        return cls.get_driver(browser, headless)


# pytest fixture 集成
@pytest.fixture(scope="session")
def driver(request):
    """pytest session 级 driver fixture"""
    browser = request.config.getoption("--browser", default="chrome")
    headless = request.config.getoption("--headless", default=False)
    
    drv = DriverManager.get_driver(browser, headless)

    DriverManager.quit_driver()
```

```python
# config/environment.py
"""多环境策略模式"""

from abc import ABC, abstractmethod
from dataclasses import dataclass


@dataclass
class EnvironmentConfig:
    """环境配置数据类"""
    base_url: str
    api_url: str
    db_host: str
    db_port: int
    db_name: str
    timeout: int
    retry_count: int
    log_level: str


class EnvironmentStrategy(ABC):
    """环境策略抽象基类"""
    
    @abstractmethod
    def get_config(self) -> EnvironmentConfig:
        pass
    
    @abstractmethod
    def get_browser_config(self) -> dict:
        pass


class DevEnvironment(EnvironmentStrategy):
    """开发环境"""
    
    def get_config(self) -> EnvironmentConfig:
        return EnvironmentConfig(
            base_url="http://localhost:3000",
            api_url="http://localhost:8080/api/v1",
            db_host="localhost",
            db_port=5432,
            db_name="test_db_dev",
            timeout=30,
            retry_count=1,
            log_level="DEBUG"
        )
    
    def get_browser_config(self) -> dict:
        return {"headless": False, "slow_mo": 100}


class StagingEnvironment(EnvironmentStrategy):
    """预发布环境"""
    
    def get_config(self) -> EnvironmentConfig:
        return EnvironmentConfig(
            base_url="https://staging.example.com",
            api_url="https://api-staging.example.com/v1",
            db_host="staging-db.internal",
            db_port=5432,
            db_name="test_db_staging",
            timeout=60,
            retry_count=2,
            log_level="INFO"
        )
    
    def get_browser_config(self) -> dict:
        return {"headless": True, "slow_mo": 0}


class ProductionEnvironment(EnvironmentStrategy):
    """生产环境（仅做冒烟测试）"""
    
    def get_config(self) -> EnvironmentConfig:
        return EnvironmentConfig(
            base_url="https://www.example.com",
            api_url="https://api.example.com/v1",
            db_host="prod-db.internal",
            db_port=5432,
            db_name="test_db_prod",
            timeout=120,
            retry_count=3,
            log_level="WARNING"
        )
    
    def get_browser_config(self) -> dict:
        return {"headless": True, "slow_mo": 0}


class EnvironmentContext:
    """环境上下文 - 运行时切换策略"""
    
    _strategies = {
        "dev": DevEnvironment,
        "staging": StagingEnvironment,
        "production": ProductionEnvironment,
    }
    
    def __init__(self, env_name: str = "dev"):
        strategy_class = self._strategies.get(env_name)
        if not strategy_class:
            raise ValueError(f"未知环境: {env_name}，可选: {list(self._strategies.keys())}")
        self._strategy = strategy_class()
    
    @property
    def config(self) -> EnvironmentConfig:
        return self._strategy.get_config()
    
    @property
    def browser_config(self) -> dict:
        return self._strategy.get_browser_config()
```

### 实操5：Decorator 装饰器模式

```python
# utils/decorators.py
"""测试装饰器库"""

import functools
import time
import logging
from typing import Callable, Any
from selenium.common.exceptions import (
    StaleElementReferenceException, 
    NoSuchElementException,
    TimeoutException
)

logger = logging.getLogger(__name__)


def retry(max_attempts: int = 3, delay: float = 1.0, exceptions: tuple = (Exception,)):
    """
    重试装饰器 - 测试中处理不稳定操作
    
    Args:
        max_attempts: 最大重试次数
        delay: 重试间隔（秒）
        exceptions: 触发重试的异常类型
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            last_exception = None
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt < max_attempts:
                        logger.warning(
                            f"{func.__name__} 第 {attempt} 次失败，{delay}s 后重试: {e}"
                        )
                        time.sleep(delay)
                    else:
                        logger.error(
                            f"{func.__name__} 重试 {max_attempts} 次后仍然失败"
                        )
            raise last_exception
        return wrapper
    return decorator


def retry_on_stale_element(max_attempts: int = 3):
    """专门针对 StaleElementReferenceException 的重试"""
    return retry(
        max_attempts=max_attempts,
        delay=0.5,
        exceptions=(StaleElementReferenceException, NoSuchElementException)
    )


def log_step(step_name: str = None):
    """
    步骤日志装饰器 - 自动记录测试步骤
    
    用法:
        @log_step("用户登录")
        def test_login():
            ...
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            name = step_name or func.__name__
            logger.info(f"=== 步骤开始: {name} ===")
            try:
                result = func(*args, **kwargs)
                logger.info(f"=== 步骤完成: {name} ===")
                return result
            except Exception as e:
                logger.error(f"=== 步骤失败: {name} | 错误: {e} ===")
                raise
        return wrapper
    return decorator


def measure_time(func: Callable) -> Callable:
    """
    耗时统计装饰器 - 记录测试步骤执行时间
    """
    @functools.wraps(func)
    def wrapper(*args, **kwargs) -> Any:
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        logger.info(f"{func.__name__} 执行耗时: {elapsed:.3f}s")
        
        # 存储到测试上下文中供报告使用
        if hasattr(wrapper, '_timing_data'):
            wrapper._timing_data.append({
                "function": func.__name__,
                "duration": elapsed,
                "timestamp": time.time()
            })
        
        return result
    
    wrapper._timing_data = []
    return wrapper


def screenshot_on_failure(driver_attr: str = "driver"):
    """
    失败自动截图装饰器
    
    Args:
        driver_attr: driver 在测试类中的属性名
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            try:
                return func(*args, **kwargs)
            except Exception:
                # 从 self 中获取 driver
                test_instance = args[0] if args else None
                if test_instance and hasattr(test_instance, driver_attr):
                    driver = getattr(test_instance, driver_attr)
                    timestamp = time.strftime("%Y%m%d_%H%M%S")
                    filename = f"failure_{func.__name__}_{timestamp}.png"
                    filepath = f"reports/screenshots/{filename}"
                    driver.save_screenshot(filepath)
                    logger.info(f"失败截图: {filepath}")
                raise
        return wrapper
    return decorator


# 使用示例
class TestOrderFlow:
    """订单流程测试 - 装饰器组合使用"""
    
    @screenshot_on_failure("driver")
    @log_step("完整下单流程")
    @measure_time
    def test_complete_order_flow(self, driver):
        """测试从商品选择到支付完成的完整流程"""
        product_page = ProductPage(driver)
        product_page.add_to_cart("iPhone 15")
        cart_page = CartPage(driver)
        cart_page.proceed_to_checkout()
    
    @screenshot_on_failure("driver")
    @retry(max_attempts=3, delay=2, exceptions=(TimeoutException,))
    def test_payment_verification(self, driver):
        """支付验证（可能因网络波动失败，需要重试）"""
        payment_page = PaymentPage(driver)
        payment_page.enter_card_info(
            number="4111111111111111",
            expiry="12/27",
            cvv="123"
        )
        payment_page.submit()
        assert payment_page.is_payment_successful()
```

## 常见坑

1. **Page Object 过大**：单页元素过多（>50个）不拆分——按业务模块拆分子对象
2. **POM 中写断言**：Page Object 应只封装操作，断言写在测试用例中
3. **Factory 数据不隔离**：测试间共享可变数据导致相互影响——每个测试独立生成数据
4. **Singleton 线程不安全**：并行测试时多个线程共享同一个 driver——使用 ThreadLocal
5. **装饰器过度叠加**：5+ 层装饰器导致调试困难——不超过 3 层
