```
project/

```

分层职责：

- **config 层**：环境变量、日志配置，与代码解耦
- **api 层**：封装原始 HTTP 调用，对外提供业务语义方法
- **tests 层**：纯断言逻辑，不包含请求细节
- **utils 层**：通用工具，可被任意层调用
- **reports 层**：测试产物，纳入 `.gitignore`

### 18.2 BaseAPI 基类设计

基类统一处理请求发送、异常捕获、日志记录，子类只需定义接口路径和参数。

```python
# api/base_api.py
import requests
import logging
from typing import Optional, Dict, Any

logger = logging.getLogger(__name__)


class BaseAPI:
    def __init__(self, base_url: str, token: Optional[str] = None):
        self.session = requests.Session()
        self.base_url = base_url.rstrip("/")
        self.session.headers.update({"Content-Type": "application/json"})
        if token:
            self.session.headers["Authorization"] = f"Bearer {token}"

    def request(
        self,
        method: str,
        path: str,
        params: Optional[Dict] = None,
        json: Optional[Dict] = None,
        data: Optional[Any] = None,
        headers: Optional[Dict] = None,
        timeout: int = 10,
    ) -> requests.Response:
        url = f"{self.base_url}/{path.lstrip('/')}"
        logger.info(f"[{method}] {url}")
        logger.debug(f"params={params}, json={json}")

        try:
            resp = self.session.request(
                method=method,
                url=url,
                params=params,
                json=json,
                data=data,
                headers=headers,
                timeout=timeout,
            )
            logger.info(f"Response: {resp.status_code} ({resp.elapsed.total_seconds():.2f}s)")
            resp.raise_for_status()
            return resp
        except requests.exceptions.Timeout:
            logger.error(f"请求超时: {url}")
            raise
        except requests.exceptions.HTTPError as e:
            logger.error(f"HTTP 错误: {e.response.status_code} - {e.response.text}")
            raise
        except requests.exceptions.ConnectionError:
            logger.error(f"连接失败: {url}")
            raise

    def get(self, path, **kwargs):
        return self.request("GET", path, **kwargs)

    def post(self, path, **kwargs):
        return self.request("POST", path, **kwargs)

    def put(self, path, **kwargs):
        return self.request("PUT", path, **kwargs)

    def delete(self, path, **kwargs):
        return self.request("DELETE", path, **kwargs)
```

业务接口子类：

```python
# api/user_api.py
from api.base_api import BaseAPI


class UserAPI(BaseAPI):
    def login(self, username: str, password: str):
        return self.post("/auth/login", json={"username": username, "password": password})

    def get_profile(self, user_id: int):
        return self.get(f"/users/{user_id}")

    def create_user(self, username: str, email: str):
        return self.post("/users", json={"username": username, "email": email})
```

import yaml
import os

def load_config(env: str = None) -> dict:
    if env is None:
        env = os.getenv("TEST_ENV", "test")
    with open("config/config.yaml", "r", encoding="utf-8") as f:

  {"username": "wrong", "password": "wrong", "expected_status": 401}

**CSV 数据源**：

```csv
username,password,expected_status
admin,admin123,200
,admin123,400
admin,,400
wrong,wrong,401
```

```python
# utils/read_data.py
import json
import csv

def read_json(filepath: str) -> list:
    with open(filepath, "r", encoding="utf-8") as f:
        return json.load(f)

def read_csv(filepath: str) -> list:
    with open(filepath, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        return [row for row in reader]
```

### 18.8 入口脚本设计

```python
# run.py
import argparse
import subprocess
import sys
from datetime import datetime


def main():
    parser = argparse.ArgumentParser(description="接口自动化测试框架")
    parser.add_argument("-e", "--env", default="test", choices=["test", "staging", "prod"])
    parser.add_argument("-m", "--marker", default="", help="pytest marker 过滤")
    parser.add_argument("--html", action="store_true", help="生成 HTML 报告")
    parser.add_argument("--allure", action="store_true", help="生成 Allure 报告")
    parser.add_argument("-v", "--verbose", action="store_true", help="详细输出")

    args = parser.parse_args()

    cmd = ["pytest", "tests/"]

    if args.marker:
        cmd.extend(["-m", args.marker])
    if args.html:
        cmd.extend(["--html=reports/report.html", "--self-contained-html"])
    if args.allure:
        cmd.extend(["--alluredir=reports/allure-results"])
    if args.verbose:
        cmd.append("-v")

    env = {"TEST_ENV": args.env}
    print(f"[{datetime.now()}] 执行环境: {args.env}")
    print(f"命令: {' '.join(cmd)}")

    result = subprocess.run(cmd, env={**__import__("os").environ, **env})
    sys.exit(result.returncode)


if __name__ == "__main__":
    main()
```

```bash
# 测试环境冒烟用例 + HTML 报告
python run.py -e test -m smoke --html

# 预发布环境全量 + Allure
python run.py -e staging --allure
```

## 常见坑

1. **BaseAPI 未处理异常透传**：基类中 `raise_for_status()` 后未捕获异常，导致测试框架无法获取有意义的错误信息。应在基类中记录响应体后再抛出。
2. **Session 作用域错误**：fixture 使用 `function` 级别创建 Session，导致每次测试都重建连接池，性能下降。应使用 `session` 级别。
3. **测试数据硬编码**：断言值直接写在测试函数中，数据变更时需修改代码。应统一外移至数据文件。
4. **日志配置未隔离环境**：测试和生产的日志输出到同一文件，排查困难。应在日志文件名中加入环境标识。
5. **报告目录未纳入 .gitignore**：`reports/` 目录应加入版本控制忽略列表，避免测试产物污染仓库。
6. **pytest 未配置 `conftest.py` 路径**：`conftest.py` 必须放在 `tests/` 目录或其子目录中，否则 fixture 不生效。
7. **YAML 中密码明文存储**：敏感信息应通过环境变量注入，YAML 中仅保留占位符或引用变量名。
