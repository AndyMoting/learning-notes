#### 练习1：实现一个函数，判断用户名是否合法

**需求：** 用户名规则为6-20位，以字母开头，仅包含字母、数字、下划线。

```python
def is_valid_username(username: str) -> bool:
    """
    验证用户名是否合法
    规则：6-20位，字母开头，仅含字母/数字/下划线
    """
    if not username:
        return False
    
    # 检查长度
    if len(username) < 6 or len(username) > 20:
        return False
    
    # 检查首字符是否为字母
    if not username[0].isalpha():
        return False
    
    # 检查所有字符合法性
    for char in username:
        if not (char.isalnum() or char == '_'):
            return False
    
    return True


# 测试用例
test_cases = [
    ("alice_01", True),
    ("Bob123", True),
    ("abc", False),          # 长度不足
    ("123abc", False),       # 首字符非字母
    ("Eve@2024", False),     # 含非法字符
    ("", False),             # 空值
    ("a_very_long_username_over_limit", False),  # 长度超出
]

for username, expected in test_cases:
    result = is_valid_username(username)
    status = "通过" if result == expected else "失败"
    print(f"用户名: {username:<30} 预期: {expected:<5} 实际: {result:<5} [{status}]")
```

#### 练习2：统计列表中各元素出现次数

**需求：** 给定一个整数列表，统计每个数字出现的次数，按次数降序输出。

```python
from collections import Counter

def count_elements(numbers: list) -> list:
    """
    统计列表中各元素出现次数，按次数降序返回
    返回格式: [(元素, 次数), ...]
    """
    counter = Counter(numbers)
    # 按次数降序排列，次数相同则按元素值升序
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items


# 测试数据
```

#### 练习3：实现FizzBuzz

**需求：** 打印1-100的数字，3的倍数打印"Fizz"，5的倍数打印"Buzz"，同时是3和5的倍数打印"FizzBuzz"。

```python
def fizzbuzz(n: int = 100) -> list:
    """
    生成FizzBuzz序列
    """
    result = []
    for i in range(1, n + 1):
        if i % 15 == 0:
            result.append("FizzBuzz")
        elif i % 3 == 0:
            result.append("Fizz")
        elif i % 5 == 0:
            result.append("Buzz")
        else:
            result.append(str(i))
    return result


# 执行并打印前20个结果
output = fizzbuzz(100)
print("FizzBuzz 前20项:")
print(", ".join(output[:20]))

print(f"\n统计: 数字={number_count}, Fizz={fizz_count}, Buzz={buzz_count}, FizzBuzz={fizzbuzz_count}")
```

**需求：** 读取`products.csv`文件，计算商品总数、平均价格、最高/最低价格商品。

**products.csv 示例内容：**

```csv

102,机械键盘,299.00,150,电子产品

107,桌面收纳盒,25.50,400,办公用品

```

```python
import csv

def analyze_products(csv_file: str) -> dict:
    """
    读取商品CSV文件，返回统计信息
    """
    products = []
    
    with open(csv_file, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row in reader:
            row['price'] = float(row['price'])
            row['stock'] = int(row['stock'])
            products.append(row)
    
    if not products:
        return {"error": "CSV文件为空或格式错误"}
    
    prices = [p['price'] for p in products]
    max_product = max(products, key=lambda x: x['price'])
    min_product = min(products, key=lambda x: x['price'])
    
    # 按类别统计
    categories = {}
    for p in products:
        cat = p['category']
        if cat not in categories:
            categories[cat] = {'count': 0, 'total_price': 0}
        categories[cat]['count'] += 1
        categories[cat]['total_price'] += p['price']
    
    return {
        'total_products': len(products),
        'avg_price': round(sum(prices) / len(prices), 2),
        'max_price_product': max_product,
        'min_price_product': min_product,
        'categories': categories
    }


# 执行分析
result = analyze_products('products.csv')

print("=== 商品数据分析报告 ===")
print(f"商品总数: {result['total_products']}")
print(f"平均价格: ¥{result['avg_price']}")
print(f"最高价商品: {result['max_price_product']['name']} (¥{result['max_price_product']['price']})")
print(f"最低价商品: {result['min_price_product']['name']} (¥{result['min_price_product']['price']})")
print(f"\n按类别统计:")
for cat, info in result['categories'].items():
    avg = round(info['total_price'] / info['count'], 2)
    print(f"  {cat}: {info['count']}件, 均价¥{avg}")
```

#### 练习6：实现Bug管理类

**需求：** 设计一个Bug类，支持创建Bug、更新状态、按严重程度筛选。

```python
from datetime import datetime
from enum import Enum

class Severity(Enum):
    CRITICAL = "致命"
    MAJOR = "严重"
    NORMAL = "一般"
    MINOR = "轻微"

class BugStatus(Enum):
    NEW = "新建"
    OPEN = "打开"
    FIXED = "已修复"
    VERIFIED = "已验证"
    CLOSED = "已关闭"
    REOPENED = "重新打开"

class Bug:
    """Bug类"""
    
    _id_counter = 0  # 类变量，用于自动生成ID
    
    def __init__(self, summary: str, severity: Severity, steps: str, 
                 expected: str, actual: str):
        Bug._id_counter += 1
        self.bug_id = f"BUG-{Bug._id_counter:04d}"
        self.summary = summary
        self.severity = severity
        self.steps = steps
        self.expected = expected
        self.actual = actual
        self.status = BugStatus.NEW
        self.created_at = datetime.now()
        self.updated_at = datetime.now()
        self.assigned_to = None
    
    def update_status(self, new_status: BugStatus, assignee: str = None):
        """更新Bug状态"""
        old_status = self.status
        self.status = new_status
        self.updated_at = datetime.now()
        if assignee:
            self.assigned_to = assignee
        print(f"[{self.bug_id}] 状态变更: {old_status.value} -> {new_status.value}")
    
    def __repr__(self):
        return (f"Bug(id={self.bug_id}, severity={self.severity.value}, "
                f"status={self.status.value}, summary={self.summary})")


class BugTracker:
    """Bug管理器"""
    
    def __init__(self):
        self.bugs = []
    
    def add_bug(self, bug: Bug):
        """添加Bug"""
        self.bugs.append(bug)
        print(f"已添加: {bug}")
    
    def get_by_severity(self, severity: Severity) -> list:
        """按严重程度筛选"""
        return [b for b in self.bugs if b.severity == severity]
    
    def get_by_status(self, status: BugStatus) -> list:
        """按状态筛选"""
        return [b for b in self.bugs if b.status == status]
    
    def summary_report(self):
        """输出统计摘要"""
        print("\n=== Bug统计摘要 ===")
        print(f"总计: {len(self.bugs)} 个Bug")
        
        print("\n按严重程度:")
        for sev in Severity:
            count = len(self.get_by_severity(sev))
            print(f"  {sev.value}: {count}")
        
        print("\n按状态:")
        for status in BugStatus:
            count = len(self.get_by_status(status))
            if count > 0:
                print(f"  {status.value}: {count}")


# 使用示例
tracker = BugTracker()

# 创建Bug

    summary="登录页面密码明文显示",
    severity=Severity.MAJOR,
    steps="1. 打开登录页\n2. 输入密码",
    expected="密码以脱敏形式显示",
    expected="密码明文显示"
)
bug1.actual = "密码明文显示"

bug2 = Bug(
    summary="购物车数量为0仍可结算",
    severity=Severity.CRITICAL,
    steps="1. 添加商品\n2. 数量改为0\n3. 点击结算",
    expected="数量为0的商品不参与结算",
    actual="金额为0的订单被提交，后端报错"
)

bug3 = Bug(
    summary="搜索特殊字符导致500错误",
    severity=Severity.MAJOR,
    steps="1. 搜索框输入' OR 1=1 --\n2. 点击搜索",
    expected="返回正常搜索结果",
    actual="页面返回500错误"
)

# 添加到管理器

tracker.add_bug(bug2)
tracker.add_bug(bug3)

# 更新状态

# 输出统计
tracker.summary_report()

# 按严重程度筛选

for bug in tracker.get_by_severity(Severity.CRITICAL):
    print(f"  {bug.bug_id}: {bug.summary}")
```

#### 练习7：实现测试用例管理类（继承应用）

**需求：** 设计TestCase基类，以及FunctionalTestCase和APITestCase子类。

```python
from enum import Enum
from datetime import datetime

class TestResult(Enum):
    PASSED = "通过"
    FAILED = "失败"
    BLOCKED = "阻塞"
    SKIPPED = "跳过"

class TestCase:
    """测试用例基类"""
    
    _id_counter = 0
    
    def __init__(self, name: str, description: str, priority: str = "中"):
        TestCase._id_counter += 1
        self.case_id = f"TC-{TestCase._id_counter:04d}"
        self.name = name
        self.description = description
        self.priority = priority
        self.result = None
        self.duration = 0
        self.error_msg = None
    
    def execute(self):
        """执行测试（子类重写）"""
        raise NotImplementedError("子类必须实现execute方法")
    
    def get_info(self) -> str:
        return f"[{self.case_id}] {self.name} (优先级: {self.priority})"


class FunctionalTestCase(TestCase):
    """功能测试用例"""
    
    def __init__(self, name: str, description: str, priority: str = "中",
                 preconditions: str = "", steps: str = "", expected: str = ""):
        super().__init__(name, description, priority)
        self.preconditions = preconditions
        self.steps = steps
        self.expected = expected
    
    def execute(self):
        """模拟执行功能测试"""
        print(f"  执行功能测试: {self.name}")
        print(f"    前置条件: {self.preconditions}")
        print(f"    测试步骤: {self.steps}")
        print(f"    预期结果: {self.expected}")
        # 模拟执行（实际中此处为真实测试逻辑）
        self.result = TestResult.PASSED
        self.duration = 1.5
        return self.result


class APITestCase(TestCase):
    """接口测试用例"""
    
    def __init__(self, name: str, description: str, priority: str = "中",
                 method: str = "GET", url: str = "", headers: dict = None,
                 body: dict = None, expected_status: int = 200):
        super().__init__(name, description, priority)
        self.method = method
        self.url = url
        self.headers = headers or {}
        self.body = body or {}
        self.expected_status = expected_status
        self.actual_status = None
        self.response_body = None
    
    def execute(self):
        """模拟执行接口测试"""
        print(f"  执行接口测试: {self.name}")
        print(f"    请求: {self.method} {self.url}")
        if self.headers:
            print(f"    请求头: {self.headers}")
        if self.body:
            print(f"    请求体: {self.body}")
        print(f"    预期状态码: {self.expected_status}")
        # 模拟执行
        self.actual_status = 200
        self.result = TestResult.PASSED if self.actual_status == self.expected_status else TestResult.FAILED
        self.duration = 0.8
        return self.result


class TestSuite:
    """测试套件"""
    
    def __init__(self, name: str):
        self.name = name
        self.cases = []
    
    def add_case(self, case: TestCase):
        self.cases.append(case)
    
    def run_all(self):
        """执行所有测试用例"""
        print(f"\n=== 执行测试套件: {self.name} ===")
        print(f"共 {len(self.cases)} 个用例\n")
        
        for case in self.cases:
            print(case.get_info())
            case.execute()
            print(f"    结果: {case.result.value} | 耗时: {case.duration}s\n")
```

#### 练习8：批量生成测试数据

**需求：** 编写脚本批量生成用户测试数据，输出为CSV文件。

```python
import csv
import random
import string
from datetime import datetime, timedelta

def generate_test_users(count: int = 50) -> list:
    """
    批量生成测试用户数据
    """
    users = []
    domains = ["test.com", "example.com", "mail.com", "qa.org"]
    
    for i in range(1, count + 1):
        # 生成用户名
        prefix = random.choice(["user", "test", "auto", "qa"])
        suffix = ''.join(random.choices(string.digits, k=4))
        username = f"{prefix}_{suffix}"
        
        # 生成邮箱
        domain = random.choice(domains)
        email = f"{username}@{domain}"
        
        # 生成手机号
        prefix_phone = random.choice(["138", "139", "150", "188", "199"])
        phone = prefix_phone + ''.join(random.choices(string.digits, k=8))
        
        # 生成注册时间（最近30天内内）
        days_ago = random.randint(0, 30)
        created_at = datetime.now() - timedelta(days=days_ago)
        
        # 随机状态
        status = random.choices(
            ["active", "inactive", "locked"],
            weights=[0.8, 0.15, 0.05],
            k=1
        )[0]
        
        users.append({
            "user_id": 10000 + i,
            "username": username,
            "email": email,
            "phone": phone,
            "status": status,
            "created_at": created_at.strftime("%Y-%m-%d %H:%M:%S")
        })
    
    return users


def save_to_csv(users: list, filename: str):
    """保存用户数据到CSV"""
    if not users:
        print("无数据可保存")
        return
    
    fieldnames = ["user_id", "username", "email", "phone", "status", "created_at"]
    
    with open(filename, 'w', newline='', encoding='utf-8') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(users)
    
    print(f"已生成 {len(users)} 条测试用户数据 -> {filename}")


# 执行
users = generate_test_users(50)
save_to_csv(users, 'test_users.csv')

    print(f"{u['user_id']:<8} {u['username']:<15} {u['email']:<25} {u['phone']:<13} {u['status']:<10} {u['created_at']}")
```

#### 练习9：接口测试自动化脚本

**需求：** 编写脚本对ShopTest API进行自动化测试，验证登录、商品列表、创建订单三个接口。

```python
import json
from datetime import datetime

# 模拟requests（实际使用时替换为import requests）
class MockResponse:
    """模拟HTTP响应"""
    def __init__(self, status_code, json_data=None, text=""):
        self.status_code = status_code
        self._json = json_data or {}
        self.text = text or json.dumps(json_data or {})
    
    def json(self):
        return self._json


def mock_request(method, url, headers=None, json=None):
    """模拟HTTP请求（实际使用requests.request）"""
    # 模拟登录接口
    if url.endswith("/api/login") and method == "POST":
        if json and json.get("username") == "admin" and json.get("password") == "admin123":
            return MockResponse(200, {"code": 0, "token": "mock_token_abc123", "user_id": 1})
        return MockResponse(401, {"code": 1001, "message": "用户名或密码错误"})
    
    # 模拟商品列表接口
    if url.endswith("/api/products") and method == "GET":
        return MockResponse(200, {
            "code": 0,
            "data": [
                {"product_id": 101, "name": "无线鼠标", "price": 89.90},
                {"product_id": 102, "name": "机械键盘", "price": 299.00},
                {"product_id": 103, "name": "蓝牙耳机", "price": 199.50},
            ]
        })
    
    # 模拟创建订单接口
    if url.endswith("/api/orders") and method == "POST":
        if headers and "Bearer" in headers.get("Authorization", ""):
            return MockResponse(201, {"code": 0, "order_id": 50001, "message": "订单创建成功"})
        return MockResponse(401, {"code": 1002, "message": "未授权"})
    
    return MockResponse(404, {"code": 404, "message": "接口不存在"})


class APITester:
    """接口测试器"""
    
    def __init__(self, base_url: str):
        self.base_url = base_url
        self.token = None
        self.results = []
    
    def login(self, username: str, password: str) -> bool:
        """登录获取Token"""
        resp = mock_request("POST", f"{self.base_url}/api/login",
                           json={"username": username, "password": password})
        
        if resp.status_code == 200 and resp.json().get("code") == 0:
            self.token = resp.json()["token"]
            print(f"  登录成功，Token: {self.token[:20]}...")
            return True
        else:
            print(f"  登录失败: {resp.json().get('message', '未知错误')}")
            return False
    
    def get_products(self) -> dict:
        """获取商品列表"""
        resp = mock_request("GET", f"{self.base_url}/api/products")
        
        if resp.status_code == 200:
            products = resp.json().get("data", [])
            print(f"  获取到 {len(products)} 个商品")
            return resp.json()
        else:
            print(f"  获取商品列表失败: {resp.status_code}")
            return {}
    
    def create_order(self, product_id: int, quantity: int) -> dict:
        """创建订单"""
        headers = {"Authorization": f"Bearer {self.token}"}
        body = {"product_id": product_id, "quantity": quantity}
        
        resp = mock_request("POST", f"{self.base_url}/api/orders",
                           headers=headers, json=body)
        
        if resp.status_code == 201:
            print(f"  订单创建成功: {resp.json().get('order_id')}")
            return resp.json()
        else:
            print(f"  订单创建失败: {resp.json().get('message', '未知错误')}")
            return {}
    
    def run_tests(self):
        """执行测试流程"""
        print("=== ShopTest API 自动化测试 ===")
        print(f"测试时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        print(f"目标地址: {self.base_url}\n")
        
        # 测试1: 登录
        print("[测试1] 用户登录")
        login_ok = self.login("admin", "admin123")
        self.results.append({"name": "用户登录", "passed": login_ok})
        
        # 测试2: 商品列表
        print("\n[测试2] 商品列表")
        products = self.get_products()
        products_ok = len(products.get("data", [])) > 0
        self.results.append({"name": "商品列表", "passed": products_ok})
        
        # 测试3: 创建订单
        print("\n[测试3] 创建订单")
        order = self.create_order(101, 2)
        order_ok = order.get("order_id") is not None
        self.results.append({"name": "创建订单", "passed": order_ok})
        
        # 测试4: 未登录创建订单（异常场景）
        print("\n[测试4] 未登录创建订单（异常场景）")
        old_token = self.token
        self.token = None
        order_no_auth = self.create_order(101, 1)
        no_auth_ok = len(order_no_auth) == 0  # 预期失败
        self.token = old_token
        self.results.append({"name": "未登录创建订单拦截", "passed": no_auth_ok})
        
        # 汇总
        print("\n=== 测试结果汇总 ===")
        passed = sum(1 for r in self.results if r["passed"])
        total = len(self.results)
        for r in self.results:
            status = "通过" if r["passed"] else "失败"
            print(f"  [{status}] {r['name']}")
        print(f"\n通过率: {passed}/{total} ({passed/total*100:.0f}%)")


# 执行测试

tester.run_tests()
```
